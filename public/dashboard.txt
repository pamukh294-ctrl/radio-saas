<script>

let dataStore = {};
let view = "radio";
let selectedCat = "All";
let favorites = JSON.parse(localStorage.getItem("fav") || "[]");
let current = null;

/* ================= LOAD ================= */
async function load(){
  const res = await fetch("/api/radio");
  const data = await res.json();

  dataStore = data.radios;

  renderCats();
  renderList();
}

/* ================= TAB ================= */
function setView(v){
  view = v;

  document.getElementById("tab_radio").classList.remove("active");
  document.getElementById("tab_fav").classList.remove("active");

  if(v === "radio"){
    document.getElementById("tab_radio").classList.add("active");
  } else {
    document.getElementById("tab_fav").classList.add("active");
  }

  renderList();
}

/* ================= CATEGORY ================= */
function renderCats(){
  const cats = ["All", ...Object.keys(dataStore)];

  const el = document.getElementById("cats");
  el.innerHTML = "";

  cats.forEach(c=>{
    const div = document.createElement("div");
    div.className = "cat" + (c===selectedCat?" active":"");
    div.innerText = c;

    div.onclick = ()=>{
      selectedCat = c;
      renderCats();
      renderList();
    };

    el.appendChild(div);
  });
}

/* ================= LIST ================= */
function renderList(){

  const list = document.getElementById("list");
  list.innerHTML = "";

  let radios = [];

  // VIEW LOGIC
  if(view === "fav"){
    radios = favorites;
  } else {
    if(selectedCat === "All"){
      Object.keys(dataStore).forEach(k=>{
        radios = radios.concat(dataStore[k]);
      });
    } else {
      radios = dataStore[selectedCat] || [];
    }
  }

  radios.forEach(r=>{

    const isFav = favorites.find(f => f.name === r.name && f.url === r.url);

    const div = document.createElement("div");
    div.className = "card";

    div.innerHTML = `
      <div>
        <div class="name">${r.name}</div>
        <div class="status">${r.status || "fav"}</div>
      </div>

      <div class="heart">
        ${isFav ? "💜" : "🤍"}
      </div>
    `;

    // PLAY
    div.onclick = ()=>play(r);

    // FAVORİ
    div.querySelector(".heart").onclick = (e)=>{
      e.stopPropagation();
      toggleFav(r);
    };

    list.appendChild(div);
  });
}

/* ================= PLAY ================= */
function play(r){

  if(r.status === "broken"){
    alert("⚠️ Çalışmıyor");
    return;
  }

  current = r;

  const audio = document.getElementById("audio");
  audio.src = r.url;
  audio.play();
}

/* ================= FAVORİ ================= */
function toggleFav(r){

  const exists = favorites.find(f => f.name === r.name && f.url === r.url);

  if(!exists){
    favorites.push(r);
  } else {
    favorites = favorites.filter(f => f.name !== r.name || f.url !== r.url);
  }

  localStorage.setItem("fav", JSON.stringify(favorites));

  renderList();
}

/* ================= INIT ================= */
load();
